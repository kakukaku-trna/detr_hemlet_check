### 步骤 2：创建蒸馏损失函数

创建文件 `models/distillation.py`：

```python
import torch
import torch.nn as nn
import torch.nn.functional as F


class DistillationLoss(nn.Module):
    """知识蒸馏损失"""

    def __init__(self, temperature=4.0):
        super().__init__()
        self.temperature = temperature
        self.kl_loss = nn.KLDivLoss(reduction='batchmean')
        self.mse_loss = nn.MSELoss()

    def forward(self, student_logits, teacher_logits, student_bbox, teacher_bbox):
        """
        Args:
            student_logits: 学生模型的分类输出 [B, Q, C]
            teacher_logits: 教师模型的分类输出 [B, Q, C]
            student_bbox: 学生模型的边界框 [B, Q, 4]
            teacher_bbox: 教师模型的边界框 [B, Q, 4]
        """
        
        # 分类损失 (KL 散度)
        student_probs = torch.log_softmax(student_logits / self.temperature, dim=-1)
        teacher_probs = torch.softmax(teacher_logits / self.temperature, dim=-1)
        cls_loss = self.kl_loss(student_probs, teacher_probs) * (self.temperature ** 2)

        # 边界框损失 (MSE)
        bbox_loss = self.mse_loss(student_bbox, teacher_bbox)

        return cls_loss, bbox_loss


class CombinedLoss(nn.Module):
    """任务损失 + 蒸馏损失"""

    def __init__(self, criterion, distill_weight=0.5, temperature=4.0):
        super().__init__()
        self.task_criterion = criterion
        self.distill_criterion = DistillationLoss(temperature=temperature)
        self.distill_weight = distill_weight

    def forward(self, outputs, targets, teacher_outputs=None):
        """
        Args:
            outputs: 学生模型输出
            targets: 真实标签
            teacher_outputs: 教师模型输出 (可选)
        """
        
        # 任务损失
        task_loss = self.task_criterion(outputs, targets)

        if teacher_outputs is None:
            return task_loss

        # 蒸馏损失
        student_cls = outputs['pred_logits']  # [B, Q, num_classes]
        teacher_cls = teacher_outputs['pred_logits']
        student_bbox = outputs['pred_boxes']   # [B, Q, 4]
        teacher_bbox = teacher_outputs['pred_boxes']

        cls_distill, bbox_distill = self.distill_criterion(
            student_cls, teacher_cls,
            student_bbox, teacher_bbox
        )

        # 组合损失
        total_loss = task_loss['loss_ce'] + task_loss['loss_bbox'] + task_loss['loss_giou']
        total_loss += self.distill_weight * (cls_distill + 0.5 * bbox_distill)

        task_loss['loss_distill'] = cls_distill + bbox_distill
        task_loss['total_loss'] = total_loss

        return task_loss
```

### 步骤 4：基准测试脚本

创建文件 `benchmark_convnext.py`：

```python
import torch
import time
from models import build_model
from util.misc import NestedTensor

def benchmark_model(backbone, dec_layers, num_queries, 
                   enc_points, dec_points, checkpoint=None, device='cuda'):
    """基准测试单个模型"""
    
    class Args:
        pass
    
    args = Args()
    args.backbone = backbone
    args.num_decoder_layers = dec_layers
    args.num_queries = num_queries
    args.enc_n_points = enc_points
    args.dec_n_points = dec_points
    args.hidden_dim = 256
    args.dim_feedforward = 1024
    args.num_heads = 8
    args.num_encoder_layers = 6
    args.num_feature_levels = 4
    args.num_classes = 80
    args.aux_loss = True
    args.with_box_refine = False
    args.two_stage = False
    args.dropout = 0.1
    args.nheads = 8
    args.lr_backbone = 0
    args.dilation = False
    args.masks = False
    
    model, _, _ = build_model(args)
    
    if checkpoint:
        model.load_state_dict(torch.load(checkpoint))
    
    model = model.to(device)
    model.eval()
    
    # 参数量
    params = sum(p.numel() for p in model.parameters()) / 1e6
    
    # 延迟测试
    torch.cuda.synchronize() if device == 'cuda' else None
    start = time.time()
    
    with torch.no_grad():
        for _ in range(100):
            dummy_input = torch.randn(1, 3, 640, 640).to(device)
            model(dummy_input)
    
    torch.cuda.synchronize() if device == 'cuda' else None
    latency = (time.time() - start) / 100
    fps = 1 / latency
    
    return {
        'params': params,
        'latency': latency,
        'fps': fps
    }

# 运行对比测试
if __name__ == '__main__':
    print("Basic latency benchmarks (without蒸馏):")
    print("=" * 60)
    
    # ConvNeXt 基线
    result = benchmark_model(
        'convnext_tiny', 4, 150, 2, 2,
        checkpoint='./checkpoints/convnext_baseline/checkpoint.pth'
    )
    
    print(f"ConvNeXt-Tiny (4层 Decoder, 150 queries):")
    print(f"  参数: {result['params']:.1f}M")
    print(f"  延迟: {result['latency']*1000:.2f}ms")
    print(f"  FPS: {result['fps']:.1f}")
```
